    \subsection*{Funkcia QuickSort}

    \begin{lstlisting}
void QuickSort(int data[],int left, int right)
{
    if(left < right)
    {
        int i = left, j = right;
        // volba pivota na prostredny prvok pola
        pivot = data[(left + right)/2 ]; 
        do
        {
            while (data[i] < pivot) i++;
            while (pivot < data[j]) j--;
            if (i <= j)
            {
                swap(data[i], data[j]);
                i++;
                j--;
            }
        }
        while (i <= j);
        QuickSort(data, left, j);
        QuickSort(data, i, right);
    }
}
    
    
    \end{lstlisting}
    
    \subsection*{Funkcia Kopa}

    \begin{lstlisting}
void Kopa(int data[], int N, int i)
{
    int naj = i;
    int lavy = 2*i + 1;
    int pravy = 2*i + 2;

    if(lavy < N && data[lavy] > data[naj])
    {
        naj = lavy;
    }
    if(pravy < N && data[pravy] > data[naj])
    {
        naj = pravy;
    }
    if(naj != i)
    {
        swap(data[i], data[naj]);
        Kopa(data, N, naj);
    }
}
    
    
    \end{lstlisting}

    \subsection*{Funkcia HeapSort}

    \begin{lstlisting}
void HeapSort(int data[], int N)
{
    for(int i = N/2 - 1; i >= 0; i--)
    {
        Kopa(data, N, i);
    }
    for(int i = N - 1; i > 0; i--)
    {
        swap(data[0], data[i]);
        Kopa(data, i, 0);
    }
}
    
    
    \end{lstlisting}
    
    \subsection*{Funkcia Check}

    \begin{lstlisting}
bool Check(int data[], int n)
{
    for(int i = 0; i < n - 2; i++)
    {
        if(data[i+1] < data[i])
            return false;
    }
    return true;
}
    
    
    \end{lstlisting}

    
    \subsection*{Funkcia main()}

    \begin{lstlisting}
int main()
{
    int n;
    for(int j = 1; j <= 10; j++)
    {
        n = j * 1000000;
        int *arr_Quick = new int[n];
        int *arr_Heap = new int[n];

        srand(time(NULL));
        for(int k=0; k < n; k++)
        {
            *(arr_Quick+j) = rand();
            *(arr_Heap+j) = *(arr_Quick+j);
        }
        double priemer_Quick = 0, priemer_Heap = 0;
        for(int i = 0; i < 3; i++)
        {
            clock_t t;
            t = clock();
            QuickSort(arr_Quick, 0, n-1);
            t = clock() - t;
            double cas_celkom = ((double)t)/CLOCKS_PER_SEC;
            priemer_Quick += cas_celkom;

            bool okQuick = Check(arr_Quick, n);

            cout<<i+1<<". meranie QUICKSORT pre velkost pola "<<n<<": "<<cas_celkom<<" s"<<endl;
            if(okQuick)
            {
                cout<<"Zoradenie OK";
            }
            else
            {
                cout<<"ERORR";
            }
            cout<<endl;
            t = clock();
            HeapSort(arr_Heap, n);
            t = clock() - t;
            cas_celkom = ((double)t)/CLOCKS_PER_SEC;
            priemer_Heap += cas_celkom;

            bool okHeap = Check(arr_Heap, n);

            cout<<i+1<<". meranie HEAPSORT pre velkost pola "<<n<<": "<<cas_celkom<<" s"<<endl;
            if(okHeap)
            {
                cout<<"Zoradenie OK";
            }
            else
            {
                cout<<"ERORR";
            }
            cout<<endl;
        }
        cout<<"Priemerny cas QUICKSORT: "<<priemer_Quick/3<<endl;
        cout<<"Priemerny cas HEAPSORT: "<<priemer_Heap/3<<endl;
        cout<<endl;
        delete []arr_Quick;
        delete []arr_Heap;
    }

    return 0;
}
